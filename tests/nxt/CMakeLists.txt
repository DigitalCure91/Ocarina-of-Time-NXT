cmake_minimum_required(VERSION 3.20)
project(NxtHeadlessTests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NXT_SANITIZE "Run headless tests with address and undefined-behavior sanitizers" OFF)
get_filename_component(NXT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NXT_SRC "${NXT_ROOT}/soh/soh/Enhancements/NxtRpg")
add_executable(nxt_tests NxtTests.cpp "${NXT_SRC}/NxtWorld.cpp" "${NXT_SRC}/NxtNavigation.cpp" "${NXT_SRC}/NxtCombat.cpp")
# Keep the actor implementation in a separate translation unit: putting the
# fixture declarations in that file could hide the linkage bug being tested.
add_executable(nxt_actor_link_tests NxtActorLinkTests.cpp "${NXT_SRC}/NxtActors.cpp")
set(NXT_COLLISION_SOURCE "${NXT_ROOT}/soh/src/code/z_collision_check.c")
add_executable(nxt_collision_tests NxtCollisionTests.cpp "${NXT_COLLISION_SOURCE}")
target_compile_definitions(nxt_collision_tests PRIVATE CVAR_PREFIX_ENHANCEMENT="gEnhancements" LOG_LEVEL_GAME_PRINTS=6)
# The real engine C file has unrelated collision/graphics routines. Keep them
# in separate sections so the fixture needs only the exercised API boundaries.
# Match the game's warning policy for legacy C, retaining C++ fixture warnings.
if(MSVC)
 target_compile_options(nxt_collision_tests PRIVATE /Gy /Gw "$<$<COMPILE_LANGUAGE:C>:/w>")
 target_link_options(nxt_collision_tests PRIVATE /OPT:REF /OPT:NOICF)
else()
 target_compile_options(nxt_collision_tests PRIVATE -ffunction-sections -fdata-sections "$<$<COMPILE_LANGUAGE:C>:-w>")
 if(APPLE)
  target_link_options(nxt_collision_tests PRIVATE -Wl,-dead_strip)
 else()
  target_link_options(nxt_collision_tests PRIVATE -Wl,--gc-sections)
 endif()
 if(NXT_SANITIZE)
  # ASan global registration would retain every unrelated function-pointer
  # table in the engine file. Heap/stack ASan and all UBSan checks remain on.
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
   set_property(SOURCE "${NXT_COLLISION_SOURCE}" APPEND PROPERTY COMPILE_OPTIONS --param=asan-globals=0)
  elseif(CMAKE_C_COMPILER_ID MATCHES "Clang")
   set_property(SOURCE "${NXT_COLLISION_SOURCE}" APPEND PROPERTY COMPILE_OPTIONS -mllvm -asan-globals=0)
  endif()
 endif()
endif()
foreach(NXT_TARGET IN ITEMS nxt_tests nxt_actor_link_tests nxt_collision_tests)
target_include_directories(${NXT_TARGET} PRIVATE "${NXT_ROOT}" "${NXT_ROOT}/soh" "${NXT_ROOT}/soh/include" "${NXT_ROOT}/soh/src" "${NXT_ROOT}/soh/assets" "${NXT_ROOT}/libultraship/include" "${NXT_ROOT}/libultraship/include/libultraship")
target_compile_definitions(${NXT_TARGET} PRIVATE F3DEX_GBI_2)
if(NXT_SANITIZE AND NOT MSVC)
 target_compile_options(${NXT_TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(${NXT_TARGET} PRIVATE -fsanitize=address,undefined)
endif()
if(MSVC)
 target_compile_options(${NXT_TARGET} PRIVATE /utf-8 /Zc:preprocessor)
endif()
endforeach()
if(MSVC)
 # Match the game's actor-translation-unit checks without changing the legacy
 # simulation fixture's existing warning policy.
 target_compile_options(nxt_actor_link_tests PRIVATE /W3 /WX /permissive-)
endif()
enable_testing()
add_test(NAME nxt_headless COMMAND nxt_tests)
add_test(NAME nxt_actor_linkage COMMAND nxt_actor_link_tests)
add_test(NAME nxt_engine_collision COMMAND nxt_collision_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 set(NXT_OVERLAY_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/NxtGameOverlayGenerated.cpp")
 add_custom_command(OUTPUT "${NXT_OVERLAY_GENERATED}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/GameOverlayTests.py"
   --source-root "${NXT_ROOT}" --emit-source "${NXT_OVERLAY_GENERATED}"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/GameOverlayTests.py"
   "${NXT_ROOT}/libultraship/src/fast/Fast3dGui.cpp" "${NXT_SRC}/NxtGui.cpp"
  VERBATIM)
 add_executable(nxt_game_overlay_tests "${NXT_OVERLAY_GENERATED}")
 if(MSVC)
  target_compile_options(nxt_game_overlay_tests PRIVATE /utf-8 /Zc:preprocessor)
 elseif(NXT_SANITIZE)
  target_compile_options(nxt_game_overlay_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nxt_game_overlay_tests PRIVATE -fsanitize=address,undefined)
 endif()
 add_test(NAME nxt_game_overlay COMMAND nxt_game_overlay_tests)
else()
 message(STATUS "Python3 was not found: nxt_game_overlay test target is unavailable")
endif()
